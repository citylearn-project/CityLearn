class CityLearnEnv(Environment, Env):
    r"""CityLearn environment: a Gymnasium environment that simulates a community of buildings.

    Each step applies the controller's actions to the buildings' storage, EV chargers and flexible
    loads for one control interval and returns the next observations and rewards. After an episode,
    :py:meth:`evaluate_v2` returns the KPI table and :py:meth:`scorecard` its default selection.

    Parameters
    ----------
    schema: Union[str, Path, Mapping[str, Any]]
        Name of CityLearn data set, filepath to JSON representation or :code:`dict` object of a CityLearn schema.
        Call :py:meth:`citylearn.data.DataSet.get_dataset_names` for the list of public CityLearn data sets.
    root_directory: Union[str, Path]
        Absolute path to directory that contains the data files including the schema.
    buildings: Union[List[Building], List[str], List[int]], optional
        Buildings to include in environment. If list of :code:`citylearn.building.Building` is provided, will override :code:`buildings` definition in schema.
        If list of :str: is provided will include only schema :code:`buildings` keys that are contained in provided list of :code:`str`.
        If list of :int: is provided will include only schema :code:`buildings` whose index is contained in provided list of :code:`int`.
    electric_vehicles: Union[List[ElectricVehicle], List[str], List[int]], optional
        Electric vehicles to include, selected from the schema's :code:`electric_vehicles_def` in the same way as
        :code:`buildings`: instances, names or indices. Defaults to every EV in the schema.
    simulation_start_time_step: int, optional
        Time step to start reading data files contents.
    simulation_end_time_step: int, optional
        Time step to end reading from data files contents.
    episode_time_steps: Union[int, List[Tuple[int, int]]], optional
        If type is `int`, it is the number of time steps in an episode. If type is `List[Tuple[int, int]]]` is provided, 
        it is a list of episode start and end time steps between `simulation_start_time_step` and `simulation_end_time_step`. 
        Defaults to (`simulation_end_time_step` - `simulation_start_time_step`) + 1. An episode of N time steps
        (states) executes N - 1 control steps. `rolling_episode_split` is ignored when `episode_time_steps` is a list.
        A list outside the simulation window raises `ValueError`. When `simulation_start_time_step` or
        `simulation_end_time_step` narrows the window and this argument is not given, the schema's list is clipped
        to the window and its episodes without an interval inside the window are dropped.
    rolling_episode_split: bool, default: False
        True if episode sequences are split such that each time step is a candidate for `episode_start_time_step` otherwise, False to split episodes in steps of `episode_time_steps`.
    random_episode_split: bool, default: False
        True if episode splits are to be selected at random during training otherwise, False to select sequentially.
    time_step_ratio_mode: str, default: 'reject'
        What to do when `seconds_per_time_step` differs from the dataset step, i.e. `time_step_ratio` != 1. CityLearn
        does not resample the data: each control step reads one data row. `'reject'` raises a `ValueError`, so
        that a run never simulates distorted time. `'compress'` keeps the CityLearn 3.6.0 behaviour: each control
        step still reads one row, so the calendar, prices and EV schedules advance one row per step, while
        energies are scaled by the ratio. Can also be set with the schema key of the same name.
    episode_split_boundary: str, default: 'shared'
        How consecutive (non-rolling) splits of an integer `episode_time_steps` meet. An episode of N data rows executes
        N - 1 control steps, because its last row is only the terminal observation. `'shared'` advances consecutive episodes
        by N - 1 rows, so the terminal row of one episode is the first row of the next and every interval is simulated
        (N must be at least 2). `'disjoint'` advances them by N rows, as CityLearn 3.6.0 and earlier did, so the interval
        between two consecutive episodes is never simulated. Can also be set with the schema key of the same name. Rolling
        and explicit list splits are not affected.
    ev_soc_carryover: bool, default: False
        Start each EV of an episode from its SOC at the end of the previous episode, when the new episode continues
        it in time: it starts on the previous episode's last row (`'shared'`) or on the row after it (`'disjoint'`),
        and the previous episode ran to its end. Sequential splits carry, and an explicit list of `episode_time_steps`
        carries when a window starts exactly there. Random and rolling splits never carry, even when a pick happens
        to follow the previous episode, and the env warns once that the setting has no effect. Only the session
        connected across the boundary keeps its charge, at the new episode's first step; every later arrival takes
        the schedule SOC, and an arrival without one keeps the EV's simulated SOC, as in one run. The first episode uses the schedule SOC, as when the setting is off. Only the EV SOC is carried: batteries, thermal storage and indoor temperatures still
        start every episode from their initial state. Off, episodes are independent; on, an episode that continues the
        previous one depends on what the previous controller did. Can also be set with the schema key of the same name.
    seconds_per_time_step: float
        Number of seconds in 1 `time_step` and must be set to >= 1. It must equal the dataset step unless
        `time_step_ratio_mode='compress'`.
    time_step_ratio: int, optional
        Ratio of the control step to the dataset step. It is inferred from the data; a value passed here must
        match the inferred ratio, otherwise a `ValueError` is raised.
    reward_function: Union[RewardFunction, str], optional
        Reward function class instance or path to function class e.g. 'citylearn.reward_function.IndependentSACReward'.
        If provided, will override :code:`reward_function` definition in schema.
    reward_function_kwargs: Mapping[str, Any], optional
        Parameters to be parsed to :py:attr:`reward_function` at intialization.
    central_agent: bool, optional
        Expect 1 central agent to control all buildings.
    shared_observations: List[str], optional
        Names of common observations across all buildings i.e. observations that have the same value irrespective of the building.
    active_observations: Union[List[str], List[List[str]]], optional
        List of observations to be made available in the buildings. Can be specified for all buildings in a :code:`List[str]` or for  
        each building independently in a :code:`List[List[str]]`. Will override the observations defined in the :code:`schema`.
    inactive_observations: Union[List[str], List[List[str]]], optional
        List of observations to be made unavailable in the buildings. Can be specified for all buildings in a :code:`List[str]` or for  
        each building independently in a :code:`List[List[str]]`. Will override the observations defined in the :code:`schema`.
    active_actions: Union[List[str], List[List[str]]], optional
        List of actions to be made available in the buildings. Can be specified for all buildings in a :code:`List[str]` or for  
        each building independently in a :code:`List[List[str]]`. Will override the actions defined in the :code:`schema`.
    inactive_actions: Union[List[str], List[List[str]]], optional
        List of actions to be made unavailable in the buildings. Can be specified for all buildings in a :code:`List[str]` or for  
        each building independently in a :code:`List[List[str]]`. Will override the actions defined in the :code:`schema`.
    simulate_power_outage: Union[bool, List[bool]]
        Whether to simulate power outages. Can be specified for all buildings as single :code:`bool` or for  
        each building independently in a :code:`List[bool]`. Will override power outage defined in the :code:`schema`.
    solar_generation: Union[bool, List[bool]]
        Wehther to allow solar generation. Can be specified for all buildings as single :code:`bool` or for  
        each building independently in a :code:`List[bool]`. Will override :code:`pv` defined in the :code:`schema`.
    random_seed: int, optional
        Pseudorandom number generator seed for repeatable results.
    random_stream_version: str, optional
        Keyword argument. How the random streams for unspecified device parameters, away-from-charger EV
        state of charge, occupant set-point interactions and stochastic power outages are derived from
        `random_seed`: ``'v2'`` (default) derives one independent stream per purpose, object and episode
        with :class:`numpy.random.SeedSequence`; ``'v1'`` reproduces CityLearn 3.6.0 and earlier. Overrides
        ``random_stream_version`` in the schema. Agents built on the environment use the same version for
        their own random streams. See :py:mod:`citylearn.random_streams`.
    offline: bool, optional
        If True, disables all network fallbacks for datasets and misc sizing data. Initialization fails with a clear
        error when required files are not available locally. Defaults to the ``CITYLEARN_OFFLINE`` environment
        variable, read at initialization, or False when it is unset.
    interface: str, optional
        Observation and action contract: ``'flat'`` (default) returns one vector per agent; ``'entity'`` returns
        tables of buildings, chargers, EVs and other assets with their edges. Overrides the schema key.
        See the flat and entity interfaces guide.
    topology_mode: str, optional
        ``'static'`` (default) keeps the members and assets of the schema for the whole episode; ``'dynamic'``
        applies the schema's ``topology_events`` (members and assets joining or leaving) and requires
        ``interface='entity'``. Overrides the schema key.
    start_date: Union[str, datetime.date], optional
        Date of the first time step in rendered and exported timestamps, e.g. ``'2024-01-01'``. Defaults to the
        schema's ``start_date``, else 2024-01-01. It does not change the simulation.

    ----------------
    render_directory: Union[str, Path], optional
        Base directory where rendering and export artifacts are stored. Relative paths are resolved from the current working directory.
    render_directory_name: str, optional
        Folder name created in the current working directory for rendering and export artifacts when ``render_directory`` is not provided.

    render_session_name: str, optional

    render_mode: str, optional

        ``'end'`` for exports performed at episode completion while still allowing manual snapshots via :meth:`render`.
    export_kpis_on_episode_end: bool, optional
        Whether to automatically export ``exported_kpis.csv`` when an episode terminates.
        If not provided, defaults to the effective rendering setting (enabled when rendering is enabled).
    export_only_final_episode: bool, optional
        Whether automatic exports should run only on the final planned episode. Defaults to ``True``.
    ev_departure_within_tolerance: float, optional
        Symmetric SOC target accuracy tolerance used by EV departure within-tolerance KPIs. Defaults to ``0.05``.
    ev_departure_service_tolerance: float, optional
        Lower SOC service tolerance used by EV departure minimum-acceptable KPIs. Defaults to ``0.05``.
    business_as_usual_ev_charge_target: str, optional
        EV charging of the native BAU reference built by ``evaluate_v2()``: ``'full'`` charges to 100%,
        ``'requested'`` to the user's requested departure SOC. Defaults to the schema value, else ``'full'``.
    physics_version: int, optional
        Physics definitions: ``2`` (default) or ``1``, which reproduces the physics of CityLearn 3.6.0. Overrides the
        schema key. The migration guide lists the corrections of version 2.
    baseline_version: int, optional
        Definition of the default KPI baseline (the ``*_without_storage`` series) and of the native BAU HVAC: ``2``
        (default) removes the battery, the thermal storage and the EV charging of the controlled run, and adds back the
        EV charging of the native BAU with the ``'full'`` target, so it is the same for every controller and has no V2G;
        its native BAU also runs the HVAC so that it delivers the dataset's own heating and cooling demand. ``1`` adds
        nothing back and its BAU sends zero to the HVAC actions, as in CityLearn 3.6.0. Overrides the schema key. See
        :py:mod:`citylearn.internal.baseline_version`.
    debug_timing: bool, optional
        Log runtime timings. Defaults to the schema value, else ``False``.
    check_observation_limits: bool, optional
        Check observations against their estimated bounds. Defaults to the schema value, else ``False``.
    physics_invariant_checks: bool, optional
        Check physical invariants at runtime. Defaults to the schema value, else ``False``.
    metrics_log_interval: int, optional
        Log runtime metrics every this many steps; ``0`` (default) disables them.
    energy_provenance: bool, optional
        Whether to keep the per-step electricity provenance ledger (``*_energy_provenance_*`` KPIs and
        provenance audits). Defaults to ``True``. Disabling it does not change any simulated quantity and
        makes ``step()`` substantially faster, e.g. for training runs that do not report provenance KPIs.
    district_electrical_service: Mapping[str, Any], optional
        Opt-in shared feeder monitoring configuration, e.g.
        ``{"enabled": True, "limits": {"import_kw": 50.0, "export_kw": 20.0}}``. Measures and reports settled
        district import and export exceedances without curtailing or altering device dispatch. Either ``import_kw``
        or ``export_kw`` may be omitted or ``None`` (unmonitored in that direction), or set to a non-negative scalar
        or dataset-step-0 anchored sequence in [kW]. Defaults to the schema value, else disabled.
    evaluation_window: Union[bool, Mapping[str, Any]], optional
        Opt-in recurring local evaluation window for ``evaluate_v2()`` and KPI exports. Disabled by default.
        Pass ``True`` or ``{'enabled': True}`` to use the ``16:00:00`` to ``19:00:00`` starter window in the
        scenario's local time, or pass ``{'enabled': True, 'start': 'HH:MM', 'end': 'HH:MM', 'timezone': ...}``

        ``evaluation_window`` in the schema.

    def _get_flat_observation_space(self) -> List[spaces.Box]:
        if self.central_agent:
            low_limit = []
            high_limit = []
            shared_observations = []

            for i, b in enumerate(self.buildings):
                for l, h, s in zip(b.observation_space.low, b.observation_space.high, b.flat_observation_order):
                    if i == 0 or s not in self.shared_observations or s not in shared_observations:
                        low_limit.append(l)
                        high_limit.append(h)

                    else:
                        pass

                    if s in self.shared_observations and s not in shared_observations:
                        shared_observations.append(s)

                    else:
                        pass

            observation_space = [spaces.Box(low=np.array(low_limit), high=np.array(high_limit), dtype=np.float32)]

        else:
            observation_space = [b.observation_space for b in self.buildings]

        return observation_space

        self._observations_cache_time_step = self.time_step

    def _sum_counterfactual_series(self, attribute: str, non_dynamics_attribute: str = None) -> np.ndarray:
        """Sum a building counterfactual series over the members that make up the control series.

        The control series (`net_electricity_consumption*`) add up the members that are active at
        each step, so a member that has left keeps its past contribution. In dynamic topology the
        counterfactual sums therefore cover every member that has been active, each masked to the
        steps at which it belonged to the district and padded with zeros to the district horizon, so
        that every row keeps its dtype. `non_dynamics_attribute` replaces `attribute` for buildings
        that are not `DynamicsBuilding`.
        """

        def series(building):
            use_attribute = non_dynamics_attribute is None or isinstance(building, DynamicsBuilding)
            return getattr(building, attribute if use_attribute else non_dynamics_attribute)

        if self.topology_mode != 'dynamic':
            rows = [series(b) for b in self.buildings]

        else:
            population = ActivePopulation(self)
            members = [(b.name, np.asarray(series(b))) for b in self._kpi_service._get_kpi_buildings()]
            length = max(len(values) for _, values in members)
            rows = []

            for name, values in members:
                # A member counts as zero before it joins and after it leaves, where its series stops.
                row = np.zeros(length, dtype=values.dtype)
                row[:len(values)] = values
                row[~population.mask(name, length, energy=True)] = 0.0
                rows.append(row)

        return self._sum_member_rows(rows)

    @staticmethod
    def _sum_member_rows(rows: List[np.ndarray]) -> np.ndarray:
        """Sum equal-length member series per step; a step is NaN only if every member is NaN there.

        Bit-identical to `pd.DataFrame(rows).sum(axis=0, min_count=1).to_numpy()` without building
        the DataFrame: the rows are promoted to the same common dtype and reduced in the same order,
        over a C-contiguous (step, member) matrix with NaN read as zero. Rows of different lengths
        raise `ValueError`, where pandas would pad them with NaN and return float64;
        `_sum_counterfactual_series` pads departed members itself.
        """

        matrix = np.column_stack(rows)
        total = np.nansum(matrix, axis=1)
        total[np.isnan(matrix).all(axis=1)] = np.nan

        return total

    def _sum_building_series(self, attribute: str) -> np.ndarray:
        """Sum a building series over the members of the district at each step.

        In dynamic topology every member that has been active contributes, masked to the steps at which
        it belonged to the district, as in `net_electricity_consumption` and `_sum_counterfactual_series`:
        a member that leaves keeps its past contribution, and a member that joins adds nothing before it
        joined, so it adds zero at the steps where it was not a member. A step is NaN only if no member
        has a value there.
        """

        if self.topology_mode == 'dynamic' and getattr(self, '_topology_service', None) is not None:
            population = ActivePopulation(self)
            series = []

            for building in self._kpi_service._get_kpi_buildings():
                values = np.asarray(getattr(building, attribute)).reshape(-1)
                if values.dtype.kind not in 'fc':
                    values = values.astype('float64')
                values = values.copy()
                values[~population.mask(building.name, len(values), energy=True)] = 0.0
                series.append(values)

        elif len(self.buildings) == 0:
            return np.array([], dtype='float64')

        else:
            series = [np.asarray(getattr(b, attribute)).reshape(-1) for b in self.buildings]

        if len(series) == 0:
            return np.array([], dtype='float64')

        length = max((len(values) for values in series), default=0)
        total = np.zeros(length, dtype='float64')
        valid = np.zeros(length, dtype=bool)

        for values in series:
            end = len(values)
            if end == 0:
                continue

            try:
                mask = ~np.isnan(values)
            except TypeError:
                values = np.asarray(values, dtype='float64').reshape(-1)
                mask = ~np.isnan(values)
            if not np.any(mask):
                continue

            target = total[:end]
            target[mask] = target[mask] + values[mask]
            valid[:end] = valid[:end] | mask

        total[~valid] = np.nan
        return total

    @staticmethod
    def _schema_declares_dynamic_topology(schema: Any) -> bool:
        if not isinstance(schema, Mapping):
            return False

        schema_mode = str(schema.get('topology_mode', '')).strip().lower()
        if schema_mode == 'dynamic':
            return True

        events = schema.get('topology_events', [])
        return isinstance(events, list) and any(isinstance(event, Mapping) for event in events)

    @staticmethod
    def _parse_non_negative_float(value: Any, *, default: float, path: str) -> float:
        if value is None:
            return float(default)

        try:
            parsed = float(value)
        except (TypeError, ValueError) as exc:
            raise ValueError(f'{path} must be a non-negative finite number.') from exc

        if not np.isfinite(parsed) or parsed < 0.0:
            raise ValueError(f'{path} must be a non-negative finite number.')

        return parsed

    @schema.setter
    def schema(self, schema: Union[str, Path, Mapping[str, Any]]):
        dataset = DataSet(offline=self.offline)

        if isinstance(schema, (str, Path)):
            schema_path = Path(schema).expanduser()

            if schema_path.is_dir():
                schema_path = schema_path / 'schema.json'

            if schema_path.is_file():
                schema = FileHandler.read_json(schema_path)
                schema['root_directory'] = str(FileHandler.resolve_schema_root(schema, schema_path))

            elif isinstance(schema, str):
                try:
                    dataset_names = dataset.get_dataset_names()
                except OfflineDataError as e:
                    raise UnknownSchemaError(str(e)) from e

                if schema in dataset_names:
                    schema = dataset.get_schema(schema)
                    schema['root_directory'] = '' if schema['root_directory'] is None else schema['root_directory']
                else:
                    if self.offline:
                        raise UnknownSchemaError(
                            f"Schema '{schema}' was not found locally while offline mode is enabled. "
                            "Provide a local schema.json file path or place the dataset under a local datasets root."
                        )

                    raise UnknownSchemaError(_unknown_dataset_message(schema, dataset_names))

            else:
                raise UnknownSchemaError(f'Schema file path does not exist: {schema_path}')

        elif isinstance(schema, dict):
            schema = deepcopy(schema)
            schema['root_directory'] = '' if schema['root_directory'] is None else schema['root_directory']
        
        else:
            raise UnknownSchemaError()
        
        self.__schema = schema

    def _maybe_log_periodic_metrics(self):
        """Lightweight periodic metrics logging for long training runs."""

        interval = max(0, int(self.metrics_log_interval))

        if interval == 0:
            return

        if self.time_step <= 0:
            return

        if self.time_step % interval != 0 and not self.terminated:
            return

        net_series = default_history(self, 'net_electricity_consumption')
        if net_series is None:
            net_series = self.net_electricity_consumption or ()
        cost_series = default_history(self, 'net_electricity_consumption_cost')
        if cost_series is None:
            cost_series = self.net_electricity_consumption_cost or ()
        emission_series = default_history(self, 'net_electricity_consumption_emission')
        if emission_series is None:
            emission_series = self.net_electricity_consumption_emission or ()

        idx = min(self.time_step - 1, len(net_series) - 1)

        if idx < 0:
            return

        LOGGER.info(
            "Episode %s Step %s/%s | net_kwh=%.5f cost=%.5f co2=%.5f",
            self.episode_tracker.episode,
            self.time_step,
            self.time_steps - 1,
            float(net_series[idx]),
            float(cost_series[idx]),
            float(emission_series[idx]),
        )

    def scorecard(self, kpis: pd.DataFrame = None, level: str = None, **evaluate_kwargs: Any) -> pd.DataFrame:
        r"""Return the default scorecard: the rows of :py:meth:`evaluate_v2` named in `SCORECARD_DEFAULT_KPIS`.

        Parameters
        ----------
        kpis: pandas.DataFrame, optional
            A table returned by :py:meth:`evaluate_v2`. When omitted, :py:meth:`evaluate_v2` is called with
            `evaluate_kwargs`.
        level: str, optional
            Keep only ``'district'`` or ``'building'`` rows.
        **evaluate_kwargs: Any
            Arguments of :py:meth:`evaluate_v2`, used only when `kpis` is omitted.

        Returns
        -------
        pandas.DataFrame
            The selected rows, with the columns of :py:meth:`evaluate_v2`, in the order of
            `SCORECARD_DEFAULT_KPIS` and then by building name. KPIs that do not apply to the scenario are absent.
        """

        if kpis is None:
            kpis = self.evaluate_v2(**evaluate_kwargs)
        elif evaluate_kwargs:
            raise TypeError('Pass either kpis or evaluate_v2() arguments to scorecard(), not both.')

        if level is not None and level not in ('district', 'building'):
            raise ValueError(f"level must be 'district', 'building' or None, not {level!r}.")

        order = {name: index for index, name in enumerate(self.SCORECARD_DEFAULT_KPIS)}
        rows = kpis[kpis['cost_function'].isin(order)]

        if level is not None:
            rows = rows[rows['level'] == level]

        return (
            rows.assign(_scorecard_order=rows['cost_function'].map(order))
            .sort_values(['_scorecard_order', 'name'], kind='stable')
            .drop(columns='_scorecard_order')
            .reset_index(drop=True)
        )

    def reset(self, seed: int = None, options: Mapping[str, Any] = None) -> Tuple[Any, dict]:
        r"""Reset `CityLearnEnv` to initial state.

        Parameters
        ----------
        seed: int, optional
            New :code:`citylearn.CityLearnEnv.random_seed`. The per-episode random streams (EV drift,
            occupant interactions, stochastic outages) restart at the key of episode 0. Parameters drawn
            when the environment was built keep their values and the episode window still advances, so
            only on a new environment built with the same seed does this repeat its first episode. Pass a
            seed to the first reset only; a seed at every reset repeats the same streams in every episode.
        options: Mapping[str, Any], optional
            Optional ``episode_index`` selects a specific episode split for replay.
        
        Returns
        -------
        observations: List[List[float]]
            :attr:`observations`.
        info: dict
            A dictionary that may contain additional information regarding the reason for a `terminated` signal.
            `info` contains auxiliary diagnostic information (helpful for debugging, learning, and logging).
            Override :meth"`get_info` to get custom key-value pairs in `info`.
        """

        # Taken before anything is reset: the EV SOC at the end of the episode that is ending.
        previous_episode_end = self._ev_soc_carryover_end_state()

        # object reset
        super().reset()
        self._final_kpis_exported = False
        if hasattr(self, '_business_as_usual_baseline_service'):
            self._business_as_usual_baseline_service.clear()

        # update seed
        if seed is not None:
            self.random_seed = seed
        else:
            pass

        # Gymnasium API: reset(seed=...) also seeds the environment's own `np_random` generator.
        Env.reset(self, seed=seed)

        # update time steps for time series
        self._episode_index_explicit = bool(options and 'episode_index' in options)
        if self._episode_index_explicit:
            self.episode_tracker.set_next_episode_index(options['episode_index'])
        self.episode_tracker.next_episode(
            self.episode_time_steps,
            self.rolling_episode_split,
            self.random_episode_split,
            self.random_seed,
            self.episode_split_boundary,
        )
        # Episode key of the per-episode streams: an explicit seed restarts them at the key of the first
        # episode of an env built with that seed (construction-time draws and the window are not reset);
        # the native BAU sidecar takes the key of the episode it evaluates.
        stream_episode = getattr(self, '_random_stream_episode_override', None)
        self.episode_tracker.random_stream_episode = (
            stream_episode if stream_episode is not None else 0 if seed is not None else self.episode_tracker.episode
        )

        if self.topology_mode == 'dynamic':
            self._topology_service.reset()

    def _ev_soc_carryover_end_state(self) -> Union[Mapping[str, Any], None]:
        """The window and the EV SOC at the last row of the current episode, if it ran to that row."""

        tracker = self.episode_tracker
        start, end = tracker.episode_start_time_step, tracker.episode_end_time_step
        if not self.ev_soc_carryover or start is None or end is None or self.time_step != end - start:
            return None

        # Devices stop at the last control step: its SOC is the one reached at the episode's last row.
        soc = {}
        for ev in list(getattr(self, '_all_electric_vehicles', None) or self.electric_vehicles or []):
            index = int(ev.battery.time_step)
            if index < len(ev.battery.soc) and np.isfinite(ev.battery.soc[index]):
                soc[ev.name] = float(ev.battery.soc[index])

        return dict(start=int(start), end=int(end), soc=soc)

    def _ev_soc_carryover_applies(self) -> bool:
        """Whether the split mode can carry EV SOC: sequential and explicit-list splits, not random or rolling ones."""

        if self.random_episode_split:
            return False
        # Rolling is ignored for explicit lists, as in `EpisodeTracker.next_episode`.
        return isinstance(self.episode_time_steps, list) or not self.rolling_episode_split

    def _ev_soc_carryover_start_after(self, end: int) -> int:
        """Row where an episode that continues one ending on row `end` starts, for the split boundary."""

        return int(end) if self.episode_split_boundary == 'shared' else int(end) + 1

    def _ev_soc_carryover_next_start(self) -> Union[int, None]:
        """First row of the episode that follows the current one when episodes are taken in order, else `None`."""

        tracker = self.episode_tracker
        start = tracker.episode_start_time_step
        if self.random_episode_split:
            return None

        if isinstance(self.episode_time_steps, list):
            splits = tracker.validate_episode_splits(self.episode_time_steps)
            index = int(tracker.episode) % len(splits)
            return int(splits[index + 1][0]) if index + 1 < len(splits) else None

        if self.rolling_episode_split:
            return None

        length = int(self.episode_time_steps)
        step = max(length - 1, 1) if self.episode_split_boundary == 'shared' else length
        next_start = int(start) + step
        return next_start if next_start + length - 1 <= tracker.simulation_end_time_step else None

    def _ev_soc_carryover_reset(self, previous_episode_end: Union[Mapping[str, Any], None]):
        """Set the EV SOC carried from the previous episode and the KPI context of the new episode.

        The context says whether the episode starts from carried SOC, which sessions continue across its
        first row, and the largest departure countdown its last step counts (see
        `CityLearnKPIService._ev_last_step_countdown`). The native BAU reference takes the evaluated env's
        context, so that it starts from the same state and counts departures the same way.
        """

        if not self.ev_soc_carryover:
            self._ev_soc_carryover_context = None
            return

        if not self._ev_soc_carryover_applies() and not getattr(self, '_ev_soc_carryover_warned', False):
            self._ev_soc_carryover_warned = True
            warnings.warn(
                'ev_soc_carryover has no effect with random or rolling episode splits: every episode starts '
                'from the schedule SOC. Use sequential splits or an explicit list of episode_time_steps.',
                UserWarning,
                stacklevel=3,
            )

        inherited = getattr(self, '_ev_soc_carryover_inherited', None)
        if inherited is not None:
            context = deepcopy(inherited)
        else:
            tracker = self.episode_tracker
            start, end = int(tracker.episode_start_time_step), int(tracker.episode_end_time_step)
            # Random and rolling splits never carry, even when a pick happens to follow the previous
            # episode, so those episodes stay independent. An explicit list carries only when the window
            # starts exactly where the previous one ends for the split boundary.
            carried = (
                self._ev_soc_carryover_applies()
                and previous_episode_end is not None
                and start == self._ev_soc_carryover_start_after(previous_episode_end['end'])
                and start > previous_episode_end['start']
            )
            continuing = []
            if carried:
                kpi = self._kpi_service
                for building in getattr(self, '_all_buildings', None) or self.buildings:
                    for charger in building.electric_vehicle_chargers or []:
                        sim = charger.charger_simulation
                        if kpi._ev_session_connected_before_window(sim):
                            continuing.append(str(sim.electric_vehicle_id[0]))
            next_start = self._ev_soc_carryover_next_start()
            context = dict(
                carried_in=bool(carried),
                soc=dict(previous_episode_end['soc']) if carried else {},
                continuing=sorted(continuing),
                # A session still connected on the next episode's first row is scored there, with the
                # carried SOC, so the last step only counts those that leave before that row.
                last_step_countdown=(
                    0.0 if next_start == end and self._ev_soc_carryover_start_after(end) == end else 1.0
                ),
            )

        self._ev_soc_carryover_context = context
        for ev in list(getattr(self, '_all_electric_vehicles', None) or self.electric_vehicles or []):
            if ev.name in context['soc']:
                ev.battery.force_set_soc(context['soc'][ev.name])

    def load_agent(self, agent: Union[str, 'Agent'] = None, **kwargs) -> Union[Any, 'Agent']:
        """Return :class:`Agent` or sub class object as defined by the `schema`.

        Parameters
        ----------
        agent: Union[str, 'citylearn.agents.base.Agent], optional
            Agent class or string describing path to agent class, e.g. 'citylearn.agents.base.NoControlAgent'.
            If a value is not provided, defaults to the agent defined in the schema:agent:type.

        **kwargs : dict
            Agent initialization attributes. For most agents e.g. CityLearn and Stable-Baselines3 agents,
            an intialized :py:attr:`env` must be parsed to the agent :py:meth:`init` function.
            When `agent` is not provided, these update the schema:agent:attributes instead of replacing them.
        
        Returns
        -------
        agent: Agent
            Initialized agent.
        """

        # set agent class
        if agent is not None:
            agent_type = agent

            if not isinstance(agent_type, str):
                agent_type = [agent_type.__module__] + [agent_type.__name__]
                agent_type = '.'.join(agent_type)

            else:
                pass

        # set agent init attributes
        else:
            agent_type = self.schema['agent']['type']

        if agent is None:
            # The schema's agent keeps its schema attributes when keyword arguments, e.g. a wrapped `env`, are passed.
            agent_attributes = {**(self.schema['agent'].get('attributes') or {}), **kwargs}

        else:
            agent_attributes = dict(kwargs)

        if 'env' not in agent_attributes:
            agent_attributes['env'] = self

        agent_module = '.'.join(agent_type.split('.')[0:-1])
        agent_name = agent_type.split('.')[-1]
        agent_constructor = getattr(importlib.import_module(agent_module), agent_name)
        agent = agent_constructor(**agent_attributes)

        return agent

    def as_dict(self) -> dict:
        """
        Convert the current simulation state to a dictionary.

        This includes key performance indicators such as energy usage, emissions, 
        and electricity pricing at the current time step.

        Returns
        -------
        dict
            Dictionary with energy and environmental metrics for the current step.
        """
        net_series = default_history(self, 'net_electricity_consumption')
        if net_series is None:
            net_series = self.net_electricity_consumption or ()
        emission_series = default_history(self, 'net_electricity_consumption_emission')
        if emission_series is None:
            emission_series = self.net_electricity_consumption_emission or ()
        cost_series = default_history(self, 'net_electricity_consumption_cost')
        if cost_series is None:
            cost_series = self.net_electricity_consumption_cost or ()

        if len(net_series) == 0:
            idx = 0
        else:
            idx = max(0, min(self.time_step, len(net_series) - 1))

        def _safe_value(series, index: int) -> float:
            return float(series[index]) if 0 <= index < len(series) else 0.0

        self_consumption = 0.0
        stored_energy = 0.0
        total_solar_generation = 0.0

        for building in self.buildings:
            self_consumption += (
                _safe_value(building.energy_from_electrical_storage, idx)
                + _safe_value(building.energy_from_cooling_storage, idx)
                + _safe_value(building.energy_from_heating_storage, idx)
                + _safe_value(building.energy_from_dhw_storage, idx)
            )
            stored_energy += _safe_value(building.energy_to_electrical_storage, idx)
            total_solar_generation += _safe_value(building.solar_generation, idx)

        return {
            "Net Electricity Consumption-kWh": _safe_value(net_series, idx),
            "Self Consumption-kWh": self_consumption,
            "Stored energy by community- kWh": stored_energy,
            "Total Solar Generation-kWh": total_solar_generation,
            "CO2-kg_co2": _safe_value(emission_series, idx),
            "Price-$": _safe_value(cost_series, idx),
        }
    

    

def _unknown_dataset_message(name: str, dataset_names: List[str]) -> str:
    """Name the closest dataset names for a schema string that is neither a dataset name nor a file."""

    import difflib

    matches = difflib.get_close_matches(name, dataset_names, n=3, cutoff=0.6)
    suggestion = f" Did you mean {' or '.join(repr(m) for m in matches)}?" if matches else ''

    return (
        f"'{name}' is neither the name of a CityLearn dataset nor an existing schema file path.{suggestion} "
        'Call citylearn.data.DataSet().get_dataset_names() for the list of dataset names, or pass the path of a '
        'schema.json file or a schema dict.'
    )
