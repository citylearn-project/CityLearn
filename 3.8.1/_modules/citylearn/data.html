    @staticmethod
    def _existing_directories(directories: Iterable[str]) -> List[str]:
        existing_directories: List[str] = []
        seen_directories = set()

        for directory in directories:
            if directory is None:
                continue

            normalized_directory = os.path.abspath(os.path.expanduser(str(directory)))

            if normalized_directory in seen_directories:
                continue

            if os.path.isdir(normalized_directory):
                seen_directories.add(normalized_directory)
                existing_directories.append(normalized_directory)

        return existing_directories

    @staticmethod
    def _dataset_input_files(schema: Mapping[str, Any], root_directory: str) -> List[str]:
        """Return the input files that constructing `CityLearnEnv` from `schema` reads, as the schema names them.

        This is the single list of the schema keys that name input files. It follows the loader
        (`citylearn.internal.loading`, `loading_inputs` and `loading_components`, which also open the
        dynamics and occupant models), the demand-response and robustness services, the community tariffs
        and the time axis. Only buildings the loader loads count: every building of a dynamic topology,
        otherwise those not set to `include: false`. Each value is resolved like the loader does, with
        `os.path.join(root_directory, value)`. Constructor arguments that load more than the schema includes
        (`buildings`, `deferrable_appliances`, `topology_mode`) can read files beyond this list.
        `tests/test_dataset_cache_recovery.py` records the files a construction opens so the list stays complete.
        """

        files = []

        def add(*values):
            files.extend(v for v in values if v)

        # CityLearnEnv requires topology_mode='dynamic' for such a schema, and the loader then loads every building.
        events = schema.get('topology_events', [])
        dynamic = str(schema.get('topology_mode', '')).strip().lower() == 'dynamic' or (
            isinstance(events, list) and any(isinstance(e, Mapping) for e in events)
        )

        for name, building in (schema.get('buildings') or {}).items():
            if not dynamic and not parse_bool(building.get('include', True), default=True, path=f'buildings.{name}.include'):
                continue

            add(building.get('energy_simulation'), building.get('weather'), building.get('carbon_intensity'), building.get('pricing'))

            if building.get('dynamics') is not None:
                add(building['dynamics'].get('attributes', {}).get('filename'))

            if building.get('occupant') is not None:
                attributes = building['occupant'].get('attributes', {})
                add(
                    building['occupant'].get('parameters_filename'),
                    attributes.get('setpoint_increase_model_filename'),
                    attributes.get('setpoint_decrease_model_filename'),
                )

            for charger in (building.get('chargers') or {}).values():
                add(charger.get('charger_simulation'))

            for appliance in (building.get('deferrable_appliances') or {}).values():
                add(appliance.get('cycle_profiles_file'), appliance.get('flexibility_schedule_file'))

            for escalator in (building.get('escalators') or {}).values():
                add(escalator.get('simulation') or escalator.get('simulation_file'))

            pv = building.get('pv')

            if pv is not None and parse_bool(pv.get('autosize', False), default=False, path=f'buildings.{name}.pv.autosize'):
                add((pv.get('autosize_attributes') or {}).get('epw_filepath'))

        for service, key in (('demand_response', 'requests_file'), ('robustness', 'events_file')):
            config = schema.get(service) or {}

            if parse_bool(config.get('enabled', False), default=False, path=f'{service}.enabled'):
                add(config.get(key))

        add(*((schema.get('community_market') or {}).get('tariff_files') or {}).values())
        calendar = (schema.get('time_axis') or {}).get('calendar_file')

        # The time axis reads the implicit Portugal calendar only when it exists, so it is never missing.
        if (
            calendar is None and str(schema.get('dataset_contract_version', '')).startswith('portugal-native')
            and os.path.isfile(os.path.join(root_directory, 'shared', 'calendar.parquet'))
        ):
            calendar = 'shared/calendar.parquet'

        add(calendar)

        return list(dict.fromkeys(files))

    @classmethod
    def _missing_dataset_files(cls, directory: Union[Path, str]) -> List[str]:
        """Return the files of the dataset in `directory` that its loading needs but that do not exist."""

        directory = str(directory)
        schema_filepath = os.path.join(directory, 'schema.json')

        if not os.path.isfile(schema_filepath):
            return ['schema.json']

        schema = FileHandler.read_json(schema_filepath)

        return [
            f for f in cls._dataset_input_files(schema, directory)
            if not os.path.isfile(os.path.join(directory, f))
        ]

    def _local_dataset_roots(self, name: str) -> List[str]:
        """Folders with a schema.json that hold dataset `name`, other than its cache folder.

        They are `name` itself when it is such a folder, then `name` under each local datasets root.
        """

        normalized_name = str(name).strip()
        cache_root = os.path.join(self.cache_directory, 'datasets', Path(normalized_name.rstrip('/')).name)
        cache_root = os.path.normcase(os.path.realpath(cache_root))
        candidates = [Path(normalized_name).expanduser()]
        candidates += [Path(root) / normalized_name for root in self.local_datasets_directories]
        roots = []

        for candidate in candidates:
            if candidate.is_dir() and (candidate / 'schema.json').is_file():
                root = str(candidate.resolve())

                if os.path.normcase(root) != cache_root and root not in roots:
                    roots.append(root)

        return roots

    def _find_local_dataset_root(self, name: str) -> Optional[str]:
        """Return the first complete local copy of dataset `name`, or None."""

        return next((r for r in self._local_dataset_roots(name) if len(self._missing_dataset_files(r)) == 0), None)

    def _find_local_misc_filepath(self, filename: str) -> Optional[str]:
        for root in self.local_misc_directories:
            filepath = Path(root) / filename

            if filepath.is_file():
                return str(filepath.resolve())

        return None

    def _get_local_dataset_names(self) -> List[str]:
        dataset_names = set()

        for root in self.local_datasets_directories:
            for dataset_path in Path(root).iterdir():
                if dataset_path.is_dir() and (dataset_path / 'schema.json').is_file():
                    dataset_names.add(dataset_path.name)

        return sorted(dataset_names)

    @staticmethod
    def _directory_signature(directory: Union[Path, str]) -> tuple:
        """Return `(relative path, size, mtime in ns)` of every file under `directory`, sorted by path."""

        directory = str(directory)
        entries = []

        for root, _, filenames in os.walk(directory, followlinks=True):
            for filename in filenames:
                filepath = os.path.join(root, filename)
                stat = os.stat(filepath)
                relative_path = os.path.relpath(filepath, directory).replace(os.sep, '/')
                entries.append((relative_path, stat.st_size, stat.st_mtime_ns))

        return tuple(sorted(entries))

    @staticmethod
    def _hash_directory(directory: Union[Path, str], signature: tuple) -> str:
        """Return the content key of the files in `signature`: a digest of their relative paths and bytes."""

        digest = hashlib.sha256()

        for relative_path, _, _ in signature:
            file_digest = hashlib.sha256()

            with open(os.path.join(str(directory), *relative_path.split('/')), 'rb') as f:
                for chunk in iter(lambda: f.read(1 << 20), b''):
                    file_digest.update(chunk)

            digest.update(relative_path.encode('utf-8') + b'\0' + file_digest.digest())

        return digest.hexdigest()[:16]

    @classmethod
    def _content_key(cls, directory: Union[Path, str]) -> str:
        """Return the content key of the local dataset folder `directory`.

        The key is a digest of the relative path and bytes of every file in the folder, so two folders get
        the same key exactly when they hold the same files. Hashing a large dataset takes a moment, so the
        key is kept for the rest of the process and computed again only when the size or modification
        time of a file changes, or a file is added or removed; a cache hit then costs one `stat` per file.
        """

        directory = os.path.realpath(str(directory))
        signature = cls._directory_signature(directory)
        known = _LOCAL_DATASET_CONTENT_KEYS.get(directory)

        if known is not None and known[0] == signature:
            return known[1]

        key = cls._hash_directory(directory, signature)
        _LOCAL_DATASET_CONTENT_KEYS[directory] = (signature, key)

        return key

    def get_dataset(self, name: str, directory: Union[Path, str] = None) -> str:
        dataset_name = Path(str(name).rstrip('/')).name
        datasets_directory = os.path.join(self.cache_directory, 'datasets')
        path = FileHandler.join_url(self.datasets_path, dataset_name)
        local_root_directory = self._find_local_dataset_root(str(name))

        if local_root_directory is not None:
            # A complete local copy, such as `data/datasets` of a checkout, is cached under the key of its
            # content, not only under the release tag. Checkouts or versions with different files then never
            # read each other's copy, even when they share the tag, and a published copy is never replaced:
            # different content goes to a new folder, so environments that read the old one are not disturbed.
            key = self._content_key(local_root_directory)
            root_directory = os.path.join(self.cache_directory, self.CONTENT_CACHE_DIRECTORY, key, dataset_name)

            if len(self._missing_dataset_files(root_directory)) > 0:
                def copy(staging: str):
                    shutil.copytree(local_root_directory, staging, dirs_exist_ok=True)
                    copied_key = self._hash_directory(staging, self._directory_signature(staging))

                    if copied_key != key:
                        raise RuntimeError(
                            f"The local dataset '{local_root_directory}' changed while it was being cached. Try again."
                        )

                self._publish_directory(root_directory, copy)

            schema_filepath = os.path.join(root_directory, 'schema.json')

        else:
            # Downloads are cached under the release tag that `cache_directory` is named after.
            root_directory = os.path.join(datasets_directory, dataset_name)
            schema_filepath = os.path.join(root_directory, 'schema.json')
            # The cached dataset is complete when its schema and every input file the schema loads exist. An
            # interrupted download by an earlier version can leave schema.json without the files it names.
            missing_files = self._missing_dataset_files(root_directory)

            if len(missing_files) > 0:
                if self.offline:
                    local_directories = self.local_datasets_directories
                    local_directories_message = ', '.join(local_directories) if len(local_directories) > 0 else '(none)'
                    cached_message = (
                        f"The cached copy at '{root_directory}' lacks {', '.join(missing_files)}."
                        if os.path.isfile(schema_filepath) else f"There is no cached schema '{schema_filepath}'."
                    )
                    incomplete_message = ''.join(
                        f" The local copy at '{r}' lacks {', '.join(self._missing_dataset_files(r))}."
                        for r in self._local_dataset_roots(str(name))
                    )
                    raise OfflineDataError(
                        f"Offline mode is enabled and no complete copy of dataset '{dataset_name}' was found locally. "
                        f"{cached_message}{incomplete_message} Checked local dataset roots: {local_directories_message}. "
                        "Pass a local schema.json file path, place the complete dataset under one of the local roots "
                        "(CITYLEARN_LOCAL_DATASETS_PATH adds one) or disable offline mode (CITYLEARN_OFFLINE=0 or "
                        "offline=False) to download it."
                    )

                LOGGER.info(
                    f'The {dataset_name} dataset does not exist or is incomplete in the cache and local roots. Will download from '
                    f'{self.github_account}/{self.repository}/tree/{self.tag} GitHub repository and write to {datasets_directory}. '
                    f'Next time DataSet.get_dataset(\'{dataset_name}\') is called, it will read '
                    'from cache unless DataSet.clear_cache is run first.'
                )
                files = self._get_github_dataset_files(path)
                self._require_parquet_engine(dataset_name, [relative_path for relative_path, _ in files])
                session = self.get_requests_session()

                def download(staging: str):
                    for relative_path, url in files:
                        self._download_file(url, os.path.join(staging, *relative_path.split('/')), session=session)

                self._publish_directory(root_directory, download)

        if directory is not None:
            # Copy into `directory/<dataset name>`, the location of the returned schema.
            target_directory = os.path.join(directory, dataset_name)
            os.makedirs(target_directory, exist_ok=True)
            shutil.copytree(root_directory, target_directory, dirs_exist_ok=True)
            schema_filepath = os.path.join(target_directory, 'schema.json')

        return schema_filepath

    def _get_github_dataset_files(self, path: str) -> List[tuple]:
        """Return `(relative path, download URL)` for every file of the dataset at `path`, subfolders included.

        The parent folder listing gives the dataset's Git tree, which one recursive Git trees request lists
        in full, so a download costs two GitHub API requests however deeply the dataset is nested. Files
        are then fetched from raw.githubusercontent.com, which does not count towards the API rate limit.
        """

        parent, _, name = path.rstrip('/').rpartition('/')
        entry = None

        if parent:
            entry = next((
                c for c in self.get_github_contents(parent)
                if c.get('type') == 'dir' and c.get('name') == name and c.get('sha')
            ), None)

        if entry is not None:
            return self._get_github_tree_files(path, entry['sha'], entry.get('path', path))

        files = []

        for c in self.get_github_contents(path):
            name = self._relative_dataset_path(c['name'], c.get('path', c['name']))

            if c['type'] == 'file':
                files.append((name, c['download_url']))

            elif c['type'] == 'dir':
                files += [
                    (f'{name}/{relative_path}', url) for relative_path, url
                    in self._get_github_tree_files(FileHandler.join_url(path, name), c['sha'], c['path'])
                ]

            else:
                raise ValueError(f"Unsupported dataset entry: {c.get('path', name)} ({c['type']})")

        return files

    def _get_github_tree_files(self, path: str, sha: str, source: str) -> List[tuple]:
        files = []

        for t in self.get_github_tree(sha):
            if t['type'] == 'blob':
                relative_path = self._relative_dataset_path(t['path'], f"{source}/{t['path']}")
                files.append((relative_path, self.get_github_raw_url(FileHandler.join_url(path, relative_path))))

            elif t['type'] != 'tree':
                raise ValueError(f"Unsupported dataset entry: {source}/{t['path']} ({t['type']})")

        return files

    def get_github_tree(self, sha: str) -> List[Mapping[str, Any]]:
        """Return every entry below the Git tree `sha`, recursively."""

        if self.offline:
            raise OfflineDataError(
                'Offline mode is enabled and GitHub access is disabled. '
                'Provide local data files or disable offline mode.'
            )

        url = FileHandler.join_url('https://api.github.com/repos/', self.github_account, self.repository, 'git', 'trees', sha)
        response = self.get_requests_session().get(url, params=dict(recursive=1), **self._github_api_kwargs())

        if response.status_code != 200:
            raise Exception(f'Unable to get response from GitHub API for endpoint: {url}.'\
                f'\rReturned status code: {response.status_code};\rContent: {response.content}')

        tree = response.json()

        if tree.get('truncated'):
            raise Exception(f'GitHub API returned a truncated tree for endpoint: {url}.')

        return tree['tree']

    @classmethod
    def _publish_directory(cls, target_directory: str, populate) -> None:
        """Fill a dataset folder in a private staging folder, then move it into place.

        Concurrent first downloads used to write into, and delete, the same cache folder. A dataset
        now appears under `target_directory` only when complete, that is when its schema and every
        input file the schema loads exist; a process that loses the race to publish keeps the complete
        copy that won. Publishing, including the replacement of an incomplete folder left by an earlier
        version, holds a lock and checks again for a complete copy first, so a complete published
        dataset is never removed. Copies of local datasets are published under their content key, so a
        complete copy is never replaced by different files either; only a damaged one, with input files
        missing, is moved aside.
        """

        parent = os.path.dirname(target_directory)
        name = os.path.basename(target_directory)
        os.makedirs(parent, exist_ok=True)
        staging = tempfile.mkdtemp(prefix=f'.{name}-', dir=parent)
        stale = None

        try:
            populate(staging)
            missing_files = cls._missing_dataset_files(staging)

            if len(missing_files) > 0:
                raise FileNotFoundError(f"Dataset {name} lacks {', '.join(missing_files)} and was not cached.")

            # One lock per dataset, held by threads and processes alike and released by the operating
            # system if its holder dies. Downloads still run in parallel; only publishing is serialised.
            with FileLock(os.path.join(parent, f'.{name}.lock')):
                if len(cls._missing_dataset_files(target_directory)) > 0:
                    if os.path.isdir(target_directory):
                        # Incomplete folder, e.g. left by an interrupted download of an earlier version. Move it
                        # aside in one step (it is either gone or untouched), delete it after the lock.
                        stale = tempfile.mkdtemp(prefix=f'.{name}-stale-', dir=parent)
                        os.rename(target_directory, os.path.join(stale, name))

                    os.rename(staging, target_directory)
                    staging = None

        finally:
            for directory in (staging, stale):
                if directory is not None:
                    shutil.rmtree(directory, ignore_errors=True)

    @staticmethod
    def _write_file(filepath: str, write) -> None:
        """Call `write` with a temporary path, then move it to `filepath`, so readers never see a partial file."""

        directory = os.path.dirname(filepath)
        os.makedirs(directory, exist_ok=True)
        descriptor, temporary = tempfile.mkstemp(prefix='.write-', dir=directory)
        os.close(descriptor)

        try:
            write(temporary)
            os.replace(temporary, filepath)

        except BaseException:
            if os.path.exists(temporary):
                os.remove(temporary)

            raise

    @classmethod
    def _copy_file(cls, source: str, filepath: str) -> None:
        """Copy under a temporary name, then move into place, so readers never see a partial file."""

        cls._write_file(filepath, lambda temporary: shutil.copyfile(source, temporary))

    def _download_file(self, url: str, filepath: str, session: requests.Session = None) -> None:
        """Download `url` to `filepath`, raising on an error response instead of caching it as data.

        The content is written under a temporary name and then moved into place, so an interrupted
        download never leaves a partial file under the cached name.
        """

        session = self.get_requests_session() if session is None else session
        response = session.get(url)
        response.raise_for_status()
        directory = os.path.dirname(filepath)
        os.makedirs(directory, exist_ok=True)
        descriptor, temporary = tempfile.mkstemp(prefix='.download-', dir=directory)

        try:
            with os.fdopen(descriptor, 'wb') as f:
                f.write(response.content)

            os.replace(temporary, filepath)

        except BaseException:
            if os.path.exists(temporary):
                os.remove(temporary)

            raise

def add_robustness_noise(values, noise_std: float, random_state=None, lower: float = None, upper: float = None,
                         zero_stays_zero: bool = False) -> np.ndarray:
    """Add Gaussian robustness noise to a time series without leaving its physical range.

    The noise is additive with standard deviation `noise_std` in the unit of the series; it is not scaled by the
    series. It is drawn exactly as for an unbounded series, so the noise of the other series is unchanged, and
    with `noise_std` <= 0 the series is returned unchanged. Where the noise-free value lies within
    [`lower`, `upper`], the noisy value is clipped to that range; values already outside it keep their noise.
    With `zero_stays_zero`, values that are zero without noise (solar quantities at night) get no noise.
    """

    base = np.array(values, dtype='float32')
    noisy = base + NoiseUtils.generate_gaussian_noise(base, noise_std, random_state=random_state)
    if noise_std <= 0:
        return noisy

    low = -np.inf if lower is None else float(lower)
    high = np.inf if upper is None else float(upper)
    inside = (base >= low) & (base <= high)
    noisy = np.where(inside, np.clip(noisy, low, high), noisy).astype('float32', copy=False)
    if zero_stays_zero:
        noisy = np.where(base == 0.0, base, noisy)

    return noisy

class TimeSeriesData:
    """Generic time series data class.
    
    Each variable is stored in full as ``_<name>`` and read as ``<name>``, the part of it between
    `start_time_step` and `end_time_step`. The windowed values are kept in the instance
    ``__dict__``, so reading a variable is a plain attribute lookup. They are refreshed when a
    variable is set or the window changes, and they are left out of pickles and copies, which
    rebuild them from the full variables.
    
    Parameters
    ----------
    variable: np.array, optional
        A generic time series variable.
    start_time_step: int, optional
        Time step to start reading variables.
    end_time_step: int, optional
         Time step to end reading variables.
    """

    def __init__(self, variable: Iterable = None, start_time_step: int = None, end_time_step: int = None):
        self.variable = variable if variable is None else np.array(variable)
        self.start_time_step = start_time_step
        self.end_time_step = end_time_step

    @staticmethod
    def _slice_variable(variable: Any, start_time_step: int, end_time_step: int):
        if isinstance(variable, np.ndarray):
            start_index = 0 if start_time_step is None else start_time_step
            end_index = variable.shape[0] if end_time_step is None else end_time_step + 1
            return variable[start_index:end_index]

        if isinstance(variable, (list, tuple)):
            start_index = 0 if start_time_step is None else start_time_step
            end_index = len(variable) if end_time_step is None else end_time_step + 1
            return variable[start_index:end_index]

        return variable

    def _refresh_windowed_variables(self):
        data = self.__dict__
        start_time_step = data.get('_start_time_step')
        end_time_step = data.get('_end_time_step')

        for key, variable in list(data.items()):
            # `_<name>` is a full variable. A name that starts with an underscore is stored as
            # `__<name>` and read through `__getattr__`.
            if key.startswith('_') and not key.startswith('__'):
                data[key[1:]] = self._slice_variable(variable, start_time_step, end_time_step)

    def __getattr__(self, name: str, start_time_step: int = None, end_time_step: int = None):
        """Returns values of the named variable within the specified time steps and
        is useful for selecting episode-specific observation."""
        
        # not the most elegant solution tbh
        try:
            variable = self.__dict__[f'_{name}']
        except KeyError:
            raise AttributeError(f'_{name}')
        explicit_start = start_time_step is not None
        explicit_end = end_time_step is not None
        start_time_step = self.start_time_step if start_time_step is None else start_time_step
        end_time_step = self.end_time_step if end_time_step is None else end_time_step
        offset = int(self.__dict__.get('_time_step_offset', 0) or 0)

        if offset != 0 and explicit_start:
            start_time_step -= offset

        if offset != 0 and explicit_end:
            end_time_step -= offset

        return self._slice_variable(variable, start_time_step, end_time_step)
        
    def __setattr__(self, name: str, value: Any):
        """Sets named variable.
        
        Variables are named with a single underscore prefix.
        """

        data = self.__dict__
        data[f'_{name}'] = value

        if name.startswith('_'):
            return

        if name in ('start_time_step', 'end_time_step'):
            self._refresh_windowed_variables()
        else:
            data[name] = self._slice_variable(value, data.get('_start_time_step'), data.get('_end_time_step'))

    def __getstate__(self) -> Dict[str, Any]:
        return {key: value for key, value in self.__dict__.items() if key.startswith('_')}

    def __setstate__(self, state: Mapping[str, Any]):
        self.__dict__.update(state)
        self._refresh_windowed_variables()

    @classmethod
    def infer_time_step_seconds(
        cls, hour: np.ndarray, minutes: np.ndarray = None, seconds: np.ndarray = None, *, seconds_per_time_step: float = None,
    ) -> Union[float, None]:
        """Infer the dataset step, in seconds, from the calendar columns of every row.

        The step is the median interval between consecutive rows, wrapped at midnight, so that an isolated
        irregular row (a typo, or a daylight-saving jump in local-time data) does not change it. Repeated
        stamps (zero intervals) are left out: they are the padding of a member that is active for part of
        a dynamic-topology window. When fewer than `TIME_STEP_INFERENCE_MIN_AGREEMENT` of the other intervals
        equal that step the rows mix cadences, and a `ValueError` is raised. Rows finer than the resolution
        of their calendar columns (e.g. 15-second rows without a `seconds` column) repeat their stamps and
        cannot be timed from them, so the step is then `seconds_per_time_step`. Returns `None` for fewer
        than two rows.
        """

        hour = np.asarray(hour)

        if len(hour) < 2:
            return None

        if seconds is not None and minutes is not None and len(seconds) > 1 and len(minutes) > 1:
            clock = hour.astype('int64')*3600 + np.asarray(minutes, dtype='int64')*60 + np.asarray(seconds, dtype='int64')
            resolution = 1

        elif minutes is not None and len(minutes) > 1:
            clock = hour.astype('int64')*3600 + np.asarray(minutes, dtype='int64')*60
            resolution = 60

        else:
            clock = hour.astype('int64')*3600
            resolution = 3600

        intervals = np.mod(np.diff(clock), 24*3600)
        changed = intervals[intervals != 0]
        fallback = None if seconds_per_time_step is None else float(seconds_per_time_step)

        if len(changed) == 0 or (len(changed) < len(intervals) and fallback is not None and fallback < resolution):
            return fallback

        step = float(np.median(changed))
        agreement = float(np.mean(changed == step))

        if agreement < cls.TIME_STEP_INFERENCE_MIN_AGREEMENT:
            irregular = np.flatnonzero((intervals != step) & (intervals != 0))[:5]
            raise ValueError(
                f'Cannot infer the dataset time step: only {agreement:.1%} of the row intervals equal the median '
                f'interval of {step:g} s. The rows mix cadences or have irregular timestamps; the first irregular '
                f'intervals end at rows {(irregular + 1).tolist()}. Fix the calendar columns, or declare the step '
                'with the schema key dataset_seconds_per_time_step.'
            )

        return step

    @staticmethod
    def arrival_soc_index(
        charger_state: Iterable[float],
        electric_vehicle_id: Iterable[str],
        estimated_soc_arrival: Iterable[float],
        time_step: int,
        electric_vehicle_session_id: Iterable[str] = None,
    ) -> Union[int, None]:
        """Row whose estimated SOC initializes an EV that connects at `time_step`.

        The runtime arrival, the KPI arrival reference and the departure audit all use this row, so
        they agree on the SOC the EV starts the session with.

        The estimate is given on the incoming (state 2) rows before the connection, usually on the
        last one (`time_step` - 1). Some sessions give it only on an earlier incoming row, e.g. the
        row one step before arrival followed by a row without an estimate. When the previous row is
        incoming or away (state 2 or 3), the latest valid estimate is taken from `time_step` - 1 back
        through the contiguous incoming rows of the same EV and session. The search stops at a row of
        another EV and at an explicit session boundary, i.e. a row whose session ID differs from the
        one of the rows after it; a missing or empty session ID is not a boundary. When these rows hold
        no estimate, and always at the first step or after a connected or empty row, the `time_step`
        row itself is used. Returns `None` when no such row holds a value in [0, 1]; the EV then keeps
        its simulated SOC.
        """

        def valid(index: int) -> bool:
            if estimated_soc_arrival is None or not 0 <= index < len(estimated_soc_arrival):
                return False

            value = estimated_soc_arrival[index]
            return isinstance(value, (float, np.floating)) and 0.0 <= value <= 1.0

        def label(values: Iterable[str], index: int) -> Union[str, None]:
            value = values[index] if values is not None and 0 <= index < len(values) else None
            value = value.strip() if isinstance(value, str) else ''
            return None if value.lower() in ('', 'nan') else value

        time_step = int(time_step)
        fallback = time_step if valid(time_step) else None
        previous_state = (
            charger_state[time_step - 1]
            if charger_state is not None and 0 < time_step <= len(charger_state) else np.nan
        )

        if previous_state not in (2, 3):
            return fallback

        index = time_step - 1
        electric_vehicle = label(electric_vehicle_id, time_step)

        if electric_vehicle is not None and label(electric_vehicle_id, index) not in (None, electric_vehicle):
            return fallback

        session = label(electric_vehicle_session_id, time_step)

        while True:
            row_session = label(electric_vehicle_session_id, index)

            if row_session is not None:
                if session not in (None, row_session):
                    return fallback

                session = row_session

            if valid(index):
                return index

            if not (
                index > 0
                and charger_state[index] == 2
                and charger_state[index - 1] == 2
                and electric_vehicle_id is not None
                and electric_vehicle_id[index - 1] == electric_vehicle_id[index]
            ):
                return fallback

            index -= 1
    


    # Units of the SOC columns of a charger file, see `_soc_series_unit`.
    _SOC_UNITS = ('percent', 'fraction')

    # SOC columns of a charger file; each has one unit.
    _SOC_COLUMNS = (
        'electric_vehicle_required_soc_departure',
        'electric_vehicle_estimated_soc_arrival',
        'electric_vehicle_current_soc',
    )

    # Margin above 1 still read as a fraction when the unit is inferred. It absorbs floating-point
    # rounding such as 1.0000000000000013 in a file written as fractions.
    _SOC_FRACTION_TOLERANCE = 1.0e-6

    @classmethod
    def _soc_series_unit(cls, *series: Iterable[float], default_soc_value: float = -0.1) -> str:
        """Unit of SOC values: ``'percent'`` if any valid value is above 1, else ``'fraction'``.

        Missing values (NaN, `default_soc_value` and negative sentinels) are ignored. The unit is
        decided once for all the values together, so a single value such as 1 (1 %) in a column written
        in percent is not read as 1.0 (100 %).
        """

        for values in series:
            if values is None:
                continue

            raw = cls._soc_float_values(values)
            valid = np.isfinite(raw) & (raw >= 0.0) & (raw != default_soc_value)

            if np.any(raw[valid] > 1.0 + cls._SOC_FRACTION_TOLERANCE):
                return 'percent'

        return 'fraction'

    @classmethod
    def _soc_fraction_values_in_percent(cls, values: Iterable[float], default_soc_value: float = -0.1) -> np.ndarray:
        """Valid values strictly between 0 and 1 of a series read as percent.

        A percent column normally has none: 1 (1 %) is its smallest non-zero value. Values such as 0.3
        suggest a fraction column that one stray value above 1 turned into percent.
        """

        raw = cls._soc_float_values(values)
        valid = np.isfinite(raw) & (raw != default_soc_value)
        return raw[valid & (raw > 0.0) & (raw < 1.0)]

    @staticmethod
    def _soc_float_values(values: Iterable[float]) -> np.ndarray:
        """SOC values as a flat float64 array; values that are not numbers become NaN.

        A numeric array, such as a column of a loaded file, is converted directly. Other values (strings,
        mixed objects, nullable columns with ``pd.NA``) go through ``pd.to_numeric``, which takes about a
        second per million values: on the 15 s datasets it made building an environment 4 s slower.
        """

        dtype = getattr(values, 'dtype', None)

        if isinstance(values, (np.ndarray, pd.Series)) and isinstance(dtype, np.dtype) and dtype.kind in 'biuf':
            return np.ravel(np.asarray(values)).astype('float64')

        raw = pd.to_numeric(pd.Series(np.ravel(np.asarray(values, dtype=object))), errors='coerce')
        return raw.to_numpy(dtype='float64')

    @classmethod
    def _resolve_soc_unit(cls, value: str = None, path: str = 'soc_unit') -> Union[str, None]:
        """Validate a declared SOC unit; ``None`` means the unit is inferred."""

        if value is None:
            return None

        unit = str(value).strip().lower()

        if unit not in cls._SOC_UNITS:
            raise ValueError(f'{path} must be one of {list(cls._SOC_UNITS)}, got {value!r}.')

        return unit

    @classmethod
    def normalize_soc_series(
        cls,
        values: Iterable[float],
        default_soc_value: float = -0.1,
        noise_std: float = 0.0,
        random_state=None,
    ) -> np.ndarray:
        """Normalize SOC inputs given as fractions or percentages to the [0, 1] range.

        The unit is decided once for the whole series: percent if any valid value is above 1, else
        fraction, so 1 in a percent series is 1 % and not 100 % (#368). NaN, `default_soc_value` and
        negative values are missing and become `default_soc_value`.
        """

        return cls._normalize_soc_series(
            values, default_soc_value=default_soc_value, noise_std=noise_std, random_state=random_state,
        )
    


    @classmethod
    def _normalize_soc_series(
        cls,
        values: Iterable[float],
        default_soc_value: float = -0.1,
        noise_std: float = 0.0,
        random_state=None,
        unit: str = None,
    ) -> np.ndarray:
        """:py:meth:`normalize_soc_series` with an explicit `unit`.

        ``'percent'`` divides every value by 100 and ``'fraction'`` keeps it. ``None`` infers the
        unit once for the whole series with :py:meth:`_soc_series_unit`.
        """

        raw = np.array(values, dtype='float32')
        raw = np.where(np.isnan(raw), default_soc_value, raw)
        normalized = np.full(raw.shape, float(default_soc_value), dtype='float32')
        valid = (raw != default_soc_value) & (raw >= 0.0)

        unit = cls._soc_series_unit(values, default_soc_value=default_soc_value) \
            if unit is None else cls._resolve_soc_unit(unit, path='unit')
        percent = valid if unit == 'percent' else np.zeros(raw.shape, dtype=bool)
        fraction = valid & ~percent

        normalized[fraction] = raw[fraction]
        normalized[percent] = raw[percent] / 100.0

        if noise_std:
            noise = NoiseUtils.generate_gaussian_noise(normalized, noise_std, random_state=random_state) / 100.0
            normalized[valid] = normalized[valid] + noise[valid]

        normalized[valid] = np.clip(normalized[valid], 0.0, 1.0)
        return normalized.astype('float32')

        for column in numeric_columns:

            'passengers_expected_15min', 'arriving_trains', 'departing_trains',
            'minutes_to_next_train',
        )
        if (frame[list(non_negative)] < 0.0).any().any():
            raise ValueError(f'{source_label} has a negative demand, train-count or time-to-train value.')

        for column in ('people_detected', 'available'):
            values = frame[column].to_numpy(dtype='float64')
            if not np.isin(values, (0.0, 1.0)).all():
                raise ValueError(f'{source_label}.{column} must contain only 0 or 1.')

        expected = frame['passengers_from_trains_15min'] + frame['background_pedestrians_15min']
        if not np.allclose(
            frame['passengers_expected_15min'].to_numpy(dtype='float64'),
            expected.to_numpy(dtype='float64'), rtol=1.0e-5, atol=1.0e-4,
        ):
            raise ValueError(
                f'{source_label}.passengers_expected_15min must equal '
                'passengers_from_trains_15min + background_pedestrians_15min.'
            )

        instance = cls()
        for column in dataframe.columns:
            values = frame[column].to_numpy(copy=False)
            if column in ('people_detected', 'available', 'arriving_trains', 'departing_trains', 'time_step'):
                values = values.astype('int32')
            elif np.issubdtype(values.dtype, np.number):
                values = values.astype('float32')
            setattr(instance, column, values)

    @classmethod
    def from_dataframes(
        cls,
        *,
        cycle_profiles: pd.DataFrame,
        flexibility_schedule: pd.DataFrame,
        source_label: str = None,
    ) -> 'DeferrableApplianceSimulation':
        source_label = source_label or 'deferrable_appliance'
        missing_profiles = cls.PROFILE_REQUIRED_COLUMNS.difference(set(cycle_profiles.columns))
        if missing_profiles:
            raise ValueError(f'{source_label}.cycle_profiles is missing columns: {sorted(missing_profiles)}.')

        missing_schedule = cls.SCHEDULE_REQUIRED_COLUMNS.difference(set(flexibility_schedule.columns))
        if missing_schedule:
            raise ValueError(f'{source_label}.flexibility_schedule is missing columns: {sorted(missing_schedule)}.')

        profiles: Dict[str, Mapping[str, Any]] = {}
        for row_index, row in cycle_profiles.iterrows():
            profile_id = str(row['profile_id']).strip()
            if profile_id == '' or profile_id.lower() == 'nan':
                raise ValueError(f'{source_label}.cycle_profiles[{row_index}].profile_id is required.')
            if profile_id in profiles:
                raise ValueError(f"{source_label}.cycle_profiles contains duplicate profile_id '{profile_id}'.")

            load_profile = cls.parse_load_profile(row['load_profile'])
            if load_profile.size == 0:
                raise ValueError(f"{source_label}.cycle_profiles profile '{profile_id}' has an empty load_profile.")

            try:
                duration_steps = int(row['duration_steps'])
            except Exception as exc:
                raise ValueError(f"{source_label}.cycle_profiles profile '{profile_id}' duration_steps must be an integer.") from exc

            if duration_steps <= 0:
                raise ValueError(f"{source_label}.cycle_profiles profile '{profile_id}' duration_steps must be > 0.")
            if duration_steps != int(load_profile.size):
                raise ValueError(
                    f"{source_label}.cycle_profiles profile '{profile_id}' duration_steps={duration_steps} "
                    f"does not match load_profile length={load_profile.size}."
                )

            try:
                total_energy = float(row['total_energy_kwh'])
            except Exception as exc:
                raise ValueError(f"{source_label}.cycle_profiles profile '{profile_id}' total_energy_kwh must be numeric.") from exc

            if not np.isfinite(total_energy) or total_energy < 0.0:
                raise ValueError(f"{source_label}.cycle_profiles profile '{profile_id}' total_energy_kwh must be finite and >= 0.")

            profile_sum = float(np.sum(load_profile))
            if abs(profile_sum - total_energy) > max(1.0e-6, 1.0e-5 * max(abs(total_energy), 1.0)):
                raise ValueError(
                    f"{source_label}.cycle_profiles profile '{profile_id}' total_energy_kwh={total_energy} "
                    f"does not match load_profile sum={profile_sum}."
                )

            profiles[profile_id] = {
                'profile_id': profile_id,
                'duration_steps': duration_steps,
                'total_energy_kwh': total_energy,
                'load_profile': load_profile.astype('float32'),
            }

        schedule = []
        seen_cycle_ids = set()
        for row_index, row in flexibility_schedule.iterrows():
            cycle_id = str(row['cycle_id']).strip()
            if cycle_id == '' or cycle_id.lower() == 'nan':
                raise ValueError(f'{source_label}.flexibility_schedule[{row_index}].cycle_id is required.')
            if cycle_id in seen_cycle_ids:
                raise ValueError(f"{source_label}.flexibility_schedule contains duplicate cycle_id '{cycle_id}'.")
            seen_cycle_ids.add(cycle_id)

            profile_id = str(row['profile_id']).strip()
            if profile_id not in profiles:
                raise ValueError(
                    f"{source_label}.flexibility_schedule cycle '{cycle_id}' references unknown profile_id '{profile_id}'."
                )
            profile = profiles[profile_id]

            try:
                earliest = int(row['earliest_start_time_step'])
                latest = int(row['latest_start_time_step'])
                deadline = int(row['deadline_time_step'])
            except Exception as exc:
                raise ValueError(
                    f"{source_label}.flexibility_schedule cycle '{cycle_id}' time-step fields must be integers."
                ) from exc

            if earliest < 0 or latest < 0 or deadline < 0:
                raise ValueError(f"{source_label}.flexibility_schedule cycle '{cycle_id}' time-step fields must be >= 0.")
            if earliest > latest:
                raise ValueError(
                    f"{source_label}.flexibility_schedule cycle '{cycle_id}' earliest_start_time_step "
                    f"cannot be greater than latest_start_time_step."
                )
            if latest + int(profile['duration_steps']) - 1 > deadline:
                raise ValueError(
                    f"{source_label}.flexibility_schedule cycle '{cycle_id}' cannot finish by deadline when "
                    f"started at latest_start_time_step."
                )

            try:
                priority = float(row['priority'])
            except Exception as exc:
                raise ValueError(f"{source_label}.flexibility_schedule cycle '{cycle_id}' priority must be numeric.") from exc
            if not np.isfinite(priority):
                raise ValueError(f"{source_label}.flexibility_schedule cycle '{cycle_id}' priority must be finite.")

            schedule.append({
                'cycle_id': cycle_id,
                'profile_id': profile_id,
                'earliest_start_time_step': earliest,
                'latest_start_time_step': latest,
                'deadline_time_step': deadline,
                'priority': float(np.clip(priority, 0.0, 1.0)),
                'must_run': parse_bool(row['must_run'], default=True, path=f'{source_label}.flexibility_schedule.{cycle_id}.must_run'),
                'duration_steps': int(profile['duration_steps']),
                'total_energy_kwh': float(profile['total_energy_kwh']),
                'load_profile': profile['load_profile'],
            })

        schedule.sort(key=lambda item: (item['earliest_start_time_step'], item['latest_start_time_step'], item['cycle_id']))
        return cls(cycle_profiles=profiles, flexibility_schedule=schedule, source_label=source_label)
